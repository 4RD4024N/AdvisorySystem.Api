// courseService.js

import api from './api'; // Axios instance

export const courseService = {
  /**
   * Öðrencinin ders programýný getirir
   * @param {number} semester - Yarýyýl numarasý (1-8)
   * @returns {Promise<Object>} Ders programý verisi
   */
  getMySchedule: async (semester) => {
    try {
      const response = await api.get(
        `/student-courses/my-schedule?semester=${semester}`
      );
      return response.data;
    } catch (error) {
      console.error('? Ders programý yüklenemedi:', error);
      throw error;
    }
  }
};

const WeeklyScheduleTable = ({ weeklySchedule }) => {
  const days = ['Pazartesi', 'Salý', 'Çarþamba', 'Perþembe', 'Cuma'];
  const timeSlots = [
    '08:00', '09:00', '10:00', '11:00', '12:00',
    '13:00', '14:00', '15:00', '16:00', '17:00'
  ];

  // Belirli gün ve saatte hangi ders var?
  const getCourseAtSlot = (day, time) => {
    if (!weeklySchedule[day]) return null;
    
    return weeklySchedule[day].find(session => 
      session.startTime === time
    );
  };

  return (
    <table>
      <thead>
      <tr>
          <th>Saat</th>
          {days.map(day => <th key={day}>{day}</th>)}
     </tr>
      </thead>
      <tbody>
        {timeSlots.map(time => (
          <tr key={time}>
            <td>{time}</td>
            {days.map(day => {
    const session = getCourseAtSlot(day, time);
        
  return (
           <td key={`${day}-${time}`}>
         {session ? (
          <div className="course-cell">
       <strong>{session.courseCode}</strong>
      <div>{session.courseName}</div>
           <div>?? {session.roomNumber}</div>
                 <div>????? {session.instructorName}</div>
         <div>{session.timeSlot}</div>
    </div>
         ) : null}
  </td>
  );
 })}
     </tr>
     ))}
    </tbody>
    </table>
  );
};

ÖNEMLÝ NOTLAR
================================================================================

1. ZAMAN FORMATI
   ? Baþlangýç ve bitiþ saatleri TAM SAAT: "08:00", "09:00", "10:00" vb.
   ? Her ders tam 60 dakika: durationMinutes her zaman 60
   ? ESKÝ FORMAT KULLANMAYIN: "08:30", "10:20" gibi saatler artýk yok

2. GÜN ÝSÝMLERÝ
 - Backend Ýngilizce gün adý döner: dayOfWeek: "Monday"
   - Backend Türkçe gün adý da döner: dayName: "Pazartesi"
   - weeklySchedule objesi TÜRKÇE key'ler kullanýr

3. RESPONSE YAPISI
   - courses[]: Her dersin TÜM OTURUMLARI ile birlikte gelir
   - weeklySchedule{}: GÜNLERE GÖRE gruplandýrýlmýþ, UI için hazýr format
   - Her iki yapý da ayný veriyi içerir, kullaným amacýnýza göre seçin

TYPESCRIPT TYPE DEFINITIONS
================================================================================

interface ScheduleResponse {
  totalCourses: number;
  completedCourses: number;
  totalCredits: number;
  totalECTS: number;
  semester: string;
  courses: CourseDetail[];
  weeklySchedule: WeeklySchedule;
}

interface CourseDetail {
  courseId: number;
  courseCode: string;
  courseName: string;
  description: string;
  credits: number;
  ects: number;
  category: string;
  sectionCode: string;
  semester: number;
  isCompleted: boolean;
  grade: number | null;
  sessions: Session[];
}

interface Session {
  scheduleId: number;
  dayOfWeek: string;
  dayOfWeekNumber: number;
  dayName: string;
  startTime: string;     // "HH:mm" format
  endTime: string;          // "HH:mm" format
  timeSlot: string;         // "HH:mm-HH:mm" format
  roomNumber: string;
  instructorName: string;
  isTheory: boolean;
  sessionType: 'Teori' | 'Uygulama';
  sessionNumber: number;
  durationMinutes: number;  // Her zaman 60
}

interface WeeklySchedule {
  Pazartesi: Session[];
  Salý: Session[];
  Çarþamba: Session[];
  Perþembe: Session[];
  Cuma: Session[];
}
